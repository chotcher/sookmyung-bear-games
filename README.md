const store = (() => {
  let mem = {};
  const ok = (() => { try { localStorage.setItem("__t","1"); localStorage.removeItem("__t"); return true; } catch(e){ return false; } })();
  return {
    get(k, d){ try { const v = ok ? localStorage.getItem("smv_"+k) : mem[k]; return v==null ? d : JSON.parse(v); } catch(e){ return d; } },
    set(k, v){ try { if(ok) localStorage.setItem("smv_"+k, JSON.stringify(v)); else mem[k]=JSON.stringify(v); } catch(e){} }
  };
})();

const shuffle = a => { a = a.slice(); for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; };
const esc = s => String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
function blankOut(text, word){
  const escRe = s => s.replace(/[.*+?^${}()|[\]\\]/g,"\\$&");
  const onesRe = s => s.includes("one's")
    ? s.split("one's").map(escRe).join("(?:one's|his|her|their|my|your|our)")
    : escRe(s);
  const re = new RegExp("\\b" + onesRe(word) + "\\w*", "i");
  if(re.test(text)) return esc(text).replace(re, '<span class="blank"></span>');
  if(word.includes(" ")){
    // A/B 자리표시자는 아무 단어나 허용, 각 단어는 어간+굴절형(-ed, -ing 등)까지 매칭
    const part = t => t.includes("one's") ? onesRe(t) + "\\w*" : escRe(t).replace(/e$/, "e?") + "\\w*";
    const toks = word.split(/\s+/).filter(t => !/^[AB]$/.test(t));   // A, B는 자리표시자이므로 제외
    const loose = new RegExp("\\b" + toks.map(part).join("(?:\\s+\\S+){0,3}?\\s+"), "i");
    if(loose.test(text)) return esc(text).replace(loose, '<span class="blank"></span>');
  }
  if(word.includes(" ")){   // 불규칙 변화(ran, laid, were 등) 대비: 핵심 단어 하나로 매칭
    const key = word.split(/\s+/).filter(t => !/^[AB]$/.test(t) && !/^(a|an|the|of|to|in|on|with|up|out|for|at|be)$/i.test(t))
                    .sort((a,b) => b.length - a.length)[0];
    if(key){
      const kre = new RegExp("\\b" + escRe(key).replace(/e$/, "e?") + "\\w*", "i");
      if(kre.test(text)) return esc(text).replace(kre, '<span class="blank"></span>');
    }
  }
  const stem = word.slice(0, Math.max(4, word.length-2));
  const re2 = new RegExp("\\b" + escRe(stem) + "\\w*", "i");
  return re2.test(text) ? esc(text).replace(re2,'<span class="blank"></span>') : esc(text) + ' <span class="blank"></span>';
}
function markWord(text, word){
  const re = new RegExp("\\b(" + word.replace(/[.*+?^${}()|[\]\\]/g,"\\$&") + "\\w*)", "i");
  return re.test(text) ? esc(text).replace(re, "<u>$1</u>") : esc(text);
}

/* ---------------- 문제 생성 ---------------- */
function qTest(words){ // 테스트: 단어 → 뜻 4지선다
  return shuffle(words).map(w => {
    const others = shuffle(ALL.filter(x => x.word !== w.word)).slice(0,3).map(x => x.mean);
    return { type:"test", w, prompt:w.word, options:shuffle([w.mean, ...others]), answer:w.mean };
  });
}
function qExample(words){ // 예문: 빈칸 + 한글 해석, 단어 5지선다
  const pool = words.filter(w => (w.col && w.col.length) || (w.ex && w.ex.length));
  return shuffle(pool).map(w => {
    const src = (w.col && w.col.length) ? w.col[0] : w.ex[0];
    const others = shuffle(pool.filter(x => x.word !== w.word)).slice(0,4).map(x => x.word);
    return { type:"ex", w, en:blankOut(src[0], w.word), ko:src[1], options:shuffle([w.word, ...others]), answer:w.word };
  });
}
function qDef(words){ // 영영 빈칸: 콜로케이션 빈칸 + 영영풀이, 단어 5지선다
  const pool = words.filter(w => w.defs && w.defs.length);
  return shuffle(pool).map(w => {
    const phrase = (w.col && w.col.length) ? w.col[0][0] : (w.ex && w.ex.length ? w.ex[0][0] : w.word);
    const others = shuffle(pool.filter(x => x.word !== w.word)).slice(0,4).map(x => x.word);
    return { type:"def", w, en:blankOut(phrase, w.word), def:w.defs[0], options:shuffle([w.word, ...others]), answer:w.word };
  });
}
function qContext(words){ // 문맥 뜻 구분: 다의어 구문 → 알맞은 뜻
  const qs = [];
  words.filter(w => w.senses && w.senses.length >= 2).forEach(w => {
    w.senses.forEach((s, si) => (s.phrases||[]).forEach(p => qs.push({
      type:"ctx", w, phrase:markWord(p, w.word),
      options:w.senses.map(x => ({pos:x.pos, mean:x.mean})), answer:si
    })));
  });
  return shuffle(qs);
}
function qSynAnt(words, n){ // 유·반의어: 두 단어 관계 판단
  const qs = [];
  words.forEach(w => {
    (w.syn||[]).forEach(s => qs.push({ type:"rel", w, a:w.word, b:s, answer:"syn" }));
    (w.ant||[]).forEach(s => qs.push({ type:"rel", w, a:w.word, b:s, answer:"ant" }));
  });
  return shuffle(qs).slice(0, n || qs.length);
}

  const der = (w.der&&w.der.length)?`<div class="tagline"><span class="lbl">파생어</span>${w.der.map(d=>`<b>${esc(d[0])}</b> ${esc(d[1])}`).join(" · ")}</div>`:"";
  return `<div class="wcard" data-w="${esc(w.word)}">
    <div class="head"><div class="idx">${idx}</div>
      <div class="wd"><div class="w">${esc(w.word)}</div><div class="m">${esc(w.mean)}</div></div>
      ${starBtn(w.word)}</div>
    <div class="body"><span class="mean-hl">${esc(w.mean)}</span>
      ${defs?`<div class="defs">${defs}</div>`:""}${pairs}${syn}${ant}${der}</div>
  </div>`;
}
function rDay(day){
  const words = VOCAB_DATA.find(d=>d.day===day).words;
  const hasDef = words.some(w=>w.defs&&w.defs.length);
  const hasCtx = words.some(w=>w.senses&&w.senses.length>=2);
  const hasRel = words.some(w=>(w.syn&&w.syn.length)||(w.ant&&w.ant.length));
  const hasEx  = words.some(w=>(w.col&&w.col.length)||(w.ex&&w.ex.length));
  const modes = [
    {id:"cards", ic:"&#x1f0cf;", nm:"카드 학습", ok:true},
    {id:"test",  ic:"&#x270e;",  nm:"테스트", ok:true},
    {id:"ex",    ic:"&#x201c;",  nm:"예문 학습", ok:hasEx},
    {id:"def",   ic:"Aa",        nm:"영영 빈칸", ok:hasDef},
    {id:"ctx",   ic:"&#x2696;",  nm:"문맥 뜻 구분", ok:hasCtx},
    {id:"rel",   ic:"&#x21c4;",  nm:"유·반의어", ok:hasRel}
  ];
  app.innerHTML = bar("DAY "+day, {name:"home"}, true) + `
    <div class="wrap">
      <div class="mode-grid">${modes.map(m=>`
        <div class="mode-btn ${m.ok?"":"dis"}" data-m="${m.id}"><div class="ic">${m.ic}</div><div class="nm">${m.nm}</div></div>`).join("")}
      </div>
      <div class="sec-label">단어 목록 · ${words.length}</div>
      ${words.map((w,i)=>wordCardHTML(w,i+1)).join("")}
      <div class="footer">${CONFIG.footer}</div>
    </div>`;
  app.querySelectorAll(".wcard .head").forEach(el => el.onclick = () => el.parentElement.classList.toggle("open"));
  const back = {name:"day", day};
  app.querySelectorAll(".mode-btn").forEach(el => el.onclick = () => {
    const m = el.dataset.m;
    if(m==="cards") return go({name:"cards", day, i:0, title:"DAY "+day+" · 카드", backTo:back});
    const map = {
      test:{qs:qTest(words), t:"테스트"},
      ex:{qs:qExample(words), t:"예문"},
      def:{qs:qDef(words), t:"영영 빈칸"},
      ctx:{qs:qContext(words), t:"문맥 뜻 구분"},
      rel:{qs:qSynAnt(words,20), t:"유·반의어"}
    }[m];
    go({name:"quiz", kind:m, qs:map.qs, i:0, right:0, wrong:0, title:"DAY "+day+" · "+map.t, backTo:back});
  });
}

/* ---------------- 카드 학습 ---------------- */
function rCards(v){
  const words = v.words || VOCAB_DATA.find(d=>d.day===v.day).words;
  const w = words[v.i];
  const pairs = [...(w.col||[]), ...(w.ex||[])].slice(0,2).map(p=>esc(p[0])+" — "+esc(p[1])).join("<br>");
  app.innerHTML = bar(v.title, v.backTo) + `
    <div class="wrap">
      <div class="progress"><i style="width:${(v.i+1)/words.length*100}%"></i></div>
      <div class="meta">${v.i+1} / ${words.length}</div>
      <div class="flash" id="flash">${starBtn(w.word)}
        <div class="fw">${esc(w.word)}</div>
        ${(w.defs&&w.defs.length)?`<div class="fd">${esc(w.defs[0])}</div>`:""}
        <div class="fm">${esc(w.mean)}</div>
        ${pairs?`<div class="fsub">${pairs}</div>`:""}
        <div class="hint">탭하여 뜻 보기</div>
      </div>
      <div class="nav2">
        <button id="prev" ${v.i===0?"disabled":""}>&#x2190; 이전</button>
        <button id="next">${v.i===words.length-1?"완료":"다음 \u2192"}</button>
      </div>
    </div>`;
  document.getElementById("flash").onclick = e => { if(!e.target.classList.contains("star")) e.currentTarget.classList.toggle("show"); };
  document.getElementById("prev").onclick = () => go({...v, words, i:v.i-1});
  document.getElementById("next").onclick = () => {
    if(v.i===words.length-1) go(v.backTo); else go({...v, words, i:v.i+1});
  };
}

/* ---------------- 퀴즈 공통 화면 ---------------- */
function rQuiz(v){
  if(!v.qs.length){ app.innerHTML = bar(v.title, v.backTo) + `<div class="empty">출제할 자료가 없습니다.</div>`; return; }
  if(v.i >= v.qs.length) return go({name:"result", from:v});
  const q = v.qs[v.i];
  let box = "", ask = "", opts = "";
  if(q.type==="test"){
    box = `<div class="qword">${esc(q.prompt)}</div>`;
    ask = `알맞은 뜻을 고르세요`;
    opts = q.options.map((o,i)=>`<button class="opt" data-i="${i}">${esc(o)}</button>`).join("");
  } else if(q.type==="ex"){
    box = `<div class="qphrase">${q.en}</div><div class="qko">${esc(q.ko)}</div>`;
    ask = `빈칸에 들어갈 단어를 고르세요`;
    opts = q.options.map((o,i)=>`<button class="opt" data-i="${i}"><span class="n">${i+1}</span><span class="en">${esc(o)}</span></button>`).join("");
  } else if(q.type==="def"){
    box = `<div class="qphrase">${q.en}</div><div class="qdef">${esc(q.def)}</div>`;
    ask = `영영풀이에 해당하는 빈칸 단어를 고르세요`;
    opts = q.options.map((o,i)=>`<button class="opt" data-i="${i}"><span class="n">${i+1}</span><span class="en">${esc(o)}</span></button>`).join("");
  } else if(q.type==="ctx"){
    box = `<div class="qphrase">${q.phrase}</div>`;
    ask = `문맥상 <b>${esc(q.w.word)}</b>의 뜻으로 알맞은 것은?`;
    opts = q.options.map((o,i)=>`<button class="opt" data-i="${i}"><span><span class="poslbl">${esc(o.pos)} 뜻</span>${esc(o.mean)}</span></button>`).join("");
  } else if(q.type==="rel"){
    box = `<div class="pairq"><span class="pw">${esc(q.a)}</span><span class="q">?</span><span class="pw">${esc(q.b)}</span></div>`;
    ask = `두 단어의 관계를 고르세요`;
    opts = `<div class="duo">
      <button class="opt" data-i="syn"><span class="sym">=</span>유의어</button>
      <button class="opt" data-i="ant"><span class="sym">&#x2194;</span>반의어</button></div>`;
  }
  app.innerHTML = bar(v.title, v.backTo) + `
    <div class="wrap">
      <div class="progress"><i style="width:${(v.i)/v.qs.length*100}%"></i></div>
      <div class="meta">${v.i+1} / ${v.qs.length} · 맞음 <b>${v.right}</b> · 틀림 <b>${v.wrong}</b></div>
      <div class="qbox">${box}</div>
      <div class="qask">${ask}</div>
      ${opts}
    </div>`;
  const btns = [...app.querySelectorAll(".opt")];
  btns.forEach(btn => btn.onclick = () => {
    btns.forEach(b => b.disabled = true);
    let pick, correct;
    if(q.type==="rel"){ pick = btn.dataset.i; correct = pick === q.answer; }
    else if(q.type==="ctx"){ pick = +btn.dataset.i; correct = pick === q.answer; }
    else { pick = q.options[+btn.dataset.i]; correct = pick === q.answer; }
    btn.classList.add(correct ? "correct" : "wrong");
    if(!correct){
      btns.forEach(b => {
        const val = q.type==="rel" ? b.dataset.i : (q.type==="ctx" ? +b.dataset.i : q.options[+b.dataset.i]);
        if(val === q.answer) b.classList.add("reveal","correct");
      });
      wrongs.add(q.w.word); saveWrongs();
    } else if(v.kind === "wrongReview"){
      wrongs.delete(q.w.word); saveWrongs();
    }
    setTimeout(() => go({...v, i:v.i+1, right:v.right+(correct?1:0), wrong:v.wrong+(correct?0:1)}), correct ? 650 : 1500);
  });
}

function rResult(v){
  const f = v.from;
  const pct = Math.round(f.right / f.qs.length * 100);
  const wrongWords = [...new Set(f.qs.filter(q => wrongs.has(q.w.word)).map(q => q.w.word))];
  app.innerHTML = bar(f.title, f.backTo) + `
    <div class="wrap">
      <div class="result">
        <div class="big">${pct}<span style="font-size:22px">점</span></div>
        <div class="sm">${pct===100 ? "완벽해요!" : pct>=80 ? "잘했어요. 틀린 단어만 다시 보면 끝!" : "틀린 단어 복습으로 한 번 더 다져요."}</div>
        <div class="stat">
          <div><b>${f.qs.length}</b><span>문제</span></div>
          <div><b style="color:var(--ok)">${f.right}</b><span>맞음</span></div>
          <div><b style="color:var(--no)">${f.wrong}</b><span>틀림</span></div>
        </div>
        <button class="gbtn" id="again">다시 풀기</button>
        ${f.wrong>0 && wrongWords.length ? `<button class="gbtn line" id="retryWrong">틀린 단어만 다시</button>` : ""}
        <button class="gbtn line" id="done">돌아가기</button>
      </div>
    </div>`;
  document.getElementById("again").onclick = () => go({...f, qs:shuffle(f.qs), i:0, right:0, wrong:0});
  document.getElementById("done").onclick = () => go(f.backTo);
  const rw = document.getElementById("retryWrong");

    const ws = wrongWords.map(byWord).filter(Boolean);
    go({name:"quiz", kind:"wrongReview", qs:qTest(ws), i:0, right:0, wrong:0, title:"틀린 단어 복습", backTo:f.backTo});
  };
}
